<script>
'use strict';
const $ = selector => document.querySelector(selector);
let selected = 'Caption';
let platform = 'All platforms';
let lastResult = '';
const languages = 'Afrikaans|Albanian|Amharic|Arabic|Armenian|Assamese|Azerbaijani|Basque|Bengali|Bosnian|Bulgarian|Burmese|Catalan|Cebuano|Chinese (Simplified)|Chinese (Traditional)|Croatian|Czech|Danish|Dari|Dutch|English|Estonian|Farsi (Persian)|Filipino|Finnish|French|Georgian|German|Greek|Gujarati|Haitian Creole|Hausa|Hawaiian|Hebrew|Hindi|Hungarian|Icelandic|Igbo|Indonesian|Irish|Italian|Japanese|Javanese|Kannada|Kazakh|Khmer|Kinyarwanda|Korean|Kurdish|Lao|Latin|Latvian|Lithuanian|Macedonian|Malagasy|Malay|Malayalam|Maltese|Maori|Marathi|Mongolian|Nepali|Norwegian|Oromo|Pashto|Polish|Portuguese|Punjabi|Romanian|Russian|Serbian|Shona|Sinhala|Slovak|Slovenian|Somali|Spanish|Swahili|Swedish|Tamil|Telugu|Thai|Tigrinya|Turkish|Ukrainian|Urdu|Uzbek|Vietnamese|Welsh|Xhosa|Yoruba|Zulu|Other (enter a language)'.split('|');
for (const language of languages) {
  const option = document.createElement('option');
  option.textContent = language;
  option.value = language;
  $('#language').append(option);
}
$('#language').value = 'English';
function status(message) { $('#status').textContent = message; }
function selectTool(name) {
  const button = [...document.querySelectorAll('#tools .tool')].find(item => item.dataset.tool === name);
  if (!button) return false;
  document.querySelectorAll('#tools .tool').forEach(item => item.classList.toggle('active', item === button));
  selected = name;
  return true;
}
$('#platforms').addEventListener('click', event => {
  const button = event.target.closest('button[data-platform]');
  if (!button) return;
  document.querySelectorAll('#platforms .platform').forEach(item => item.classList.toggle('selected', item === button));
  platform = button.dataset.platform;
  status('Platform selected: ' + platform);
});
$('#tools').addEventListener('click', event => {
  const button = event.target.closest('button[data-tool]');
  if (!button) return;
  selectTool(button.dataset.tool);
  status('Tool selected: ' + selected);
});
$('#language').addEventListener('change', () => {
  $('#customLanguage').style.display = $('#language').value.startsWith('Other') ? 'block' : 'none';
});
function goToWorkspace() { $('#workspace').scrollIntoView({behavior:'smooth',block:'start'}); }
document.querySelectorAll('.nav button[data-action]').forEach(button => {
  button.addEventListener('click', () => {
    document.querySelectorAll('.nav button[data-action]').forEach(item => item.classList.toggle('active', item === button));
    const action = button.dataset.action;
    if (action === 'dashboard') window.scrollTo({top:0,behavior:'smooth'});
    if (action === 'create') { goToWorkspace(); $('#topic').focus(); }
    if (action === 'ideas') { selectTool('Content ideas'); goToWorkspace(); }
    if (action === 'rewrite') { selectTool('Description'); goToWorkspace(); status('Describe the text you want rewritten, then generate a new version.'); }
    if (action === 'translate') { goToWorkspace(); $('#language').focus(); status('Choose your target language and describe what to translate.'); }
    if (action === 'saved') { $('#result').scrollIntoView({behavior:'smooth'}); status(lastResult ? 'Your latest result is shown.' : 'No saved result yet.'); }
  });
});
$('#generate').addEventListener('click', async () => {
  const topic = $('#topic').value.trim();
  if (!topic) { status('Add a topic or video description first.'); $('#topic').focus(); return; }
  const language = $('#language').value.startsWith('Other') ? ($('#customLanguage').value.trim() || 'English') : $('#language').value;
  const tone = $('#tone').value;
  const button = $('#generate');
  button.disabled = true;
  button.textContent = '✦ Creating with AI…';
  $('#result').textContent = 'QREVIA is creating your content…';
  status('Generating…');
  try {
    const response = await fetch('/.netlify/functions/generate', {
      method:'POST', headers:{'Content-Type':'application/json'},
      body:JSON.stringify({topic,tool:selected,platform,language,tone})
    });
    const data = await response.json().catch(() => ({}));
    if (!response.ok) throw new Error(data.error || 'The AI service could not complete the request.');
    const generated = data.result || data.text;
    if (!generated) throw new Error('The AI returned an empty result.');
    lastResult = `${selected} · ${platform} · ${language} · ${tone}\n\n${generated}\n\n— QREVIA Studio`;
    $('#result').textContent = lastResult;
    status('Your content is ready!');
  } catch (error) {
    $('#result').textContent = 'We could not generate content.\n\n' + error.message;
    status('AI generation failed.');
  } finally {
    button.disabled = false;
    button.textContent = '✦ Generate content';
  }
});
$('#copy').addEventListener('click', async () => {
  if (!lastResult) { status('Nothing to copy yet.'); return; }
  try { await navigator.clipboard.writeText(lastResult); status('Copied!'); }
  catch { status('Copy unavailable. Select the result text manually.'); }
});
$('#clear').addEventListener('click', () => {
  lastResult = '';
  $('#result').textContent = 'Ready when you are. Describe your idea and choose Generate.';
  status('Result cleared.');
});
$('#logout').addEventListener('click', () => {
  sessionStorage.removeItem('qrevia_logged_in');
  window.location.href = '/login.html';
});
</script>
